from django.db import models
from django.contrib.auth.models import User

class Category(models.Model):
    name = models.CharField(max_length=100)
    description = models.TextField(blank=True)

    def __str__(self):
        return self.name

class Component(models.Model):
    name = models.CharField(max_length=255)
    category = models.ForeignKey(Category, on_delete=models.CASCADE)
    stock_level = models.IntegerField(default=0)
    min_threshold = models.IntegerField(default=5)
    specs = models.JSONField(default=dict)

    @property
    def is_low_stock(self):
        return self.stock_level <= self.min_threshold

class AllocationLog(models.Model):
    component = models.ForeignKey(Component, on_delete=models.CASCADE)
    action_type = models.CharField(max_length=50) # 'ADD', 'ALLOCATION', 'RETURN'
    quantity_changed = models.IntegerField()
    user_identity = models.CharField(max_length=150)
    timestamp = models.DateTimeField(auto_now_add=True)
